利用ws使得内外网通信
## 服务端
```go
package main

import (
	"encoding/json"
	"fmt"
	"net/http"
	"sync"
	"time"

	"github.com/gorilla/websocket"
)

const requestTimeout = 30 * time.Second

var (
	upgrade = websocket.Upgrader{}

	connMu       sync.Mutex
	internalConn *websocket.Conn

	pendingMu sync.Mutex
	pending   = make(map[string]chan []byte)
)

type tunnelRequest struct {
	ID       string `json:"id"`
	APIPath  string `json:"api_path"`
	PublicID string `json:"public_id"`
}

type tunnelResponse struct {
	ID         string `json:"id"`
	StatusCode int    `json:"status_code"`
	Body       string `json:"body"`
	Error      string `json:"error,omitempty"`
}

func wsHandler(w http.ResponseWriter, r *http.Request) {
	conn, err := upgrade.Upgrade(w, r, nil)
	if err != nil {
		return
	}

	connMu.Lock()
	if internalConn != nil {
		internalConn.Close()
	}
	internalConn = conn
	connMu.Unlock()

	fmt.Println("内网服务器已连接隧道")

	defer func() {
		connMu.Lock()
		if internalConn == conn {
			internalConn = nil
		}
		connMu.Unlock()
		conn.Close()
		fmt.Println("内网服务器已断开隧道")
	}()

	for {
		_, message, err := conn.ReadMessage()
		if err != nil {
			return
		}

		var resp tunnelResponse
		if err := json.Unmarshal(message, &resp); err != nil {
			fmt.Println("解析内网响应失败:", err)
			continue
		}

		pendingMu.Lock()
		ch, ok := pending[resp.ID]
		if ok {
			delete(pending, resp.ID)
		}
		pendingMu.Unlock()

		if ok {
			ch <- message
			close(ch)
		}
	}
}

func proxyHandler(publicID, apiPath string) http.HandlerFunc {
	return func(w http.ResponseWriter, r *http.Request) {
		if r.Method != http.MethodGet {
			w.Header().Set("Allow", http.MethodGet)
			http.Error(w, "method not allowed", http.StatusMethodNotAllowed)
			return
		}

		connMu.Lock()
		conn := internalConn
		connMu.Unlock()

		if conn == nil {
			http.Error(w, "内网隧道未建立", http.StatusServiceUnavailable)
			return
		}

		id := fmt.Sprintf("%d", time.Now().UnixNano())
		respChan := make(chan []byte, 1)

		pendingMu.Lock()
		pending[id] = respChan
		pendingMu.Unlock()

		req := tunnelRequest{
			ID:       id,
			APIPath:  apiPath,
			PublicID: publicID,
		}
		payload, err := json.Marshal(req)
		if err != nil {
			removePending(id)
			http.Error(w, "创建隧道请求失败", http.StatusInternalServerError)
			return
		}

		connMu.Lock()
		err = conn.WriteMessage(websocket.TextMessage, payload)
		connMu.Unlock()
		if err != nil {
			removePending(id)
			http.Error(w, "发送指令失败", http.StatusBadGateway)
			return
		}

		select {
		case data := <-respChan:
			var resp tunnelResponse
			if err := json.Unmarshal(data, &resp); err != nil {
				http.Error(w, "解析内网响应失败", http.StatusBadGateway)
				return
			}
			if resp.Error != "" {
				http.Error(w, resp.Error, http.StatusBadGateway)
				return
			}

			w.Header().Set("Content-Type", "application/json")
			w.WriteHeader(resp.StatusCode)
			w.Write([]byte(resp.Body))
		case <-time.After(requestTimeout):
			removePending(id)
			http.Error(w, "等待内网响应超时", http.StatusGatewayTimeout)
		}
	}
}

func removePending(id string) {
	pendingMu.Lock()
	delete(pending, id)
	pendingMu.Unlock()
}

func main() {
	http.HandleFunc("/tunnel", wsHandler)
	http.HandleFunc("/api/onchain_data/raw-signal", proxyHandler("raw-signal", "/api/onchain_data/raw-signal"))
	http.HandleFunc("/api/metrics/today-signal", proxyHandler("today-signal", "/api/metrics/today-signal"))

	fmt.Println("公网中转服务启动在 :59655...")
	http.ListenAndServe(":59655", nil)
}
```
## 客户端
```go
package main

import (
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"time"

	"github.com/gorilla/websocket"
)

const (
	serverURL   = "ws://公网IP:59655/tunnel"
	internalAPI = "http://192.168.0.132:36472"
)

var allowedPaths = map[string]struct{}{
	"/api/onchain_data/raw-signal": {},
	"/api/metrics/today-signal":    {},
}

type tunnelRequest struct {
	ID       string `json:"id"`
	APIPath  string `json:"api_path"`
	PublicID string `json:"public_id"`
}

type tunnelResponse struct {
	ID         string `json:"id"`
	StatusCode int    `json:"status_code"`
	Body       string `json:"body"`
	Error      string `json:"error,omitempty"`
}

func main() {
	for {
		conn, _, err := websocket.DefaultDialer.Dial(serverURL, nil)
		if err != nil {
			fmt.Println("连接公网失败，5秒后重连:", err)
			time.Sleep(5 * time.Second)
			continue
		}

		fmt.Println("已成功连接公网服务器")
		handleConnection(conn)
		conn.Close()
	}
}

func handleConnection(conn *websocket.Conn) {
	for {
		_, msg, err := conn.ReadMessage()
		if err != nil {
			return
		}

		var req tunnelRequest
		if err := json.Unmarshal(msg, &req); err != nil {
			writeTunnelResponse(conn, tunnelResponse{
				Error: "Invalid tunnel request",
			})
			continue
		}

		resp := fetchInternalAPI(req)
		writeTunnelResponse(conn, resp)
	}
}

func fetchInternalAPI(req tunnelRequest) tunnelResponse {
	resp := tunnelResponse{ID: req.ID}

	if _, ok := allowedPaths[req.APIPath]; !ok {
		resp.Error = "Unsupported API path"
		return resp
	}

	httpResp, err := http.Get(internalAPI + req.APIPath)
	if err != nil {
		resp.Error = "Local API Error"
		return resp
	}
	defer httpResp.Body.Close()

	body, err := io.ReadAll(httpResp.Body)
	if err != nil {
		resp.Error = "Read Local API Error"
		return resp
	}

	resp.StatusCode = httpResp.StatusCode
	resp.Body = string(body)
	return resp
}

func writeTunnelResponse(conn *websocket.Conn, resp tunnelResponse) {
	payload, err := json.Marshal(resp)
	if err != nil {
		fmt.Println("序列化响应失败:", err)
		return
	}

	if err := conn.WriteMessage(websocket.TextMessage, payload); err != nil {
		fmt.Println("发送响应失败:", err)
	}
}
```
